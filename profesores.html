// REGISTRAR NUEVO USUARIO MEDIANTE RPC document.getElementById('newUserForm').addEventListener('submit', async (e) => { e.preventDefault(); const btn = document.getElementById('btnSaveUser'); btn.disabled = true; btn.textContent = 'Creando cuenta...'; const fullName = document.getElementById('inputFullName').value.trim(); const username = document.getElementById('inputUsername').value.trim().toLowerCase(); const email = document.getElementById('inputEmail').value.trim().toLowerCase(); const role = document.getElementById('inputRole').value; const phone = document.getElementById('inputPhone').value.trim(); const password = document.getElementById('inputPassword').value; try { // Llamada a la función segura sin activar la protección CAPTCHA de signUp público const { data, error } = await supabaseClient.rpc('admin_create_user', { p_email: email, p_password: password, p_full_name: fullName, p_username: username, p_role: role, p_phone: phone }); if (error) throw error; showModalAlert('¡Usuario creado con éxito!', false); setTimeout(() => { closeModal(); loadUsers(); }, 1200); } catch (err) { showModalAlert(err.message || 'Error al crear usuario.'); } finally { btn.disabled = false; btn.textContent = 'Crear Cuenta'; } });
